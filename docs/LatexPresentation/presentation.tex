% ======================================================
\begin{frame}{Research question}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      A network is trained on tasks one after another. Each new task changes the weights.

      \bigskip
      \textbf{Does the final checkpoint still carry a measurable trace of the task learned most recently?}

      \bigskip
      Concretely: after training on $A \to B \to C$, can we tell that $C$ came last, using only the final model and models trained on a single task?
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{tikzpicture}
        \node[box, minimum width=1.6cm] (a) at (0,0) {Train\\on \textbf{A}};
        \node[box, minimum width=1.6cm] (b) at (2.1,0) {Fine-tune\\on \textbf{B}};
        \node[box, minimum width=1.6cm] (c) at (4.2,0) {Fine-tune\\on \textbf{C}};
        \draw[arr] (a)--(b); \draw[arr] (b)--(c);

        \node[font=\small, color=accent, left=3mm of f, align=right] {Which task\\came last?};
      \end{tikzpicture}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Setup: three small classification tasks}
  Each task is a 5-class problem built from CIFAR-100 classes. All tasks share one ResNet18 backbone; every task has its own 5-class head.

  \bigskip
  \centering
  \begin{tabular}{@{}llp{7.5cm}@{}}
    \toprule
    \textbf{Task} & \textbf{Theme} & \textbf{Classes}\\
    \midrule
    \hl{A} & Aquatic animals & beaver, dolphin, otter, seal, whale\\[2pt]
    \hl{B} & Electronic / office devices & clock, keyboard, lamp, telephone, television\\[2pt]
    \hl{C} & Vehicles & bicycle, bus, motorcycle, pickup truck, train\\
    \bottomrule
  \end{tabular}

  \bigskip
  \raggedright
  Six task orders are tested: all permutations of A, B, C. \\
  The \hl{actual last task} of an order is simply its final element.
\end{frame}

% ======================================================
\begin{frame}{How the sequential models are trained}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      For an order such as $A \to B \to C$:
      \begin{enumerate}
        \item Train backbone and A-head on A.
        \item Freeze the A-head, add a B-head, train on B.
        \item Freeze A- and B-heads, add a C-head, train on C.
      \end{enumerate}
      \medskip
      No forgetting protection (no EWC, SI or replay). A checkpoint is saved after every stage; the last one is the \hl{final sequential model}.
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{tikzpicture}[x=1cm,y=1cm]
        \foreach \s/\y in {1/2.4, 2/1.2, 3/0} {

\begin{frame}{The fingerprint test}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Reference models:} three models, each trained only on A, B or C (Single-A, Single-B, Single-C).

      \medskip
      \textbf{Test:} compare the final sequential model with each reference, and pick the reference that looks most similar. That task is the \hl{predicted last task}.

      \medskip
      A prediction is correct if it equals the last element of the order.

      \medskip
      Random guessing gives $1/3 = 33.3\%$.

\begin{frame}{Five families of comparison}
  \footnotesize
  \centering
  \renewcommand{\arraystretch}{1.15}
  \begin{tabular}{@{}p{3.3cm}p{5.2cm}p{5.0cm}@{}}
    \toprule
    \textbf{Method} & \textbf{What is compared} & \textbf{Prediction rule}\\
    \midrule
    Weight distance (L2) & Parameters of final model vs.\ each reference & Smallest distance\\
    CKA & Feature structure on a fixed image set & Highest similarity\\
    Feature drift & Feature vectors, sample by sample & Smallest drift\\
    Weight matching & Parameters after removing neuron reordering; direction of weight change & Smallest matched L2; highest cosine\\
    Fisher trace & Curvature of the final model only (no references) & Rank the three tasks; test both directions\\
    \bottomrule
  \end{tabular}

  \vspace{2mm}
  \raggedright\small
  Forgetting, loss barriers and Jacobian sensitivity are computed too, but only as supporting diagnostics.
\end{frame}

% ======================================================
\begin{frame}{Weight distance}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      Euclidean distance between the final model and each reference:
      \[
        D_{\text{full}} = \lVert \theta_{\text{seq}} - \theta_{\text{ref}} \rVert_2
      \]
      Two versions:
      \begin{itemize}
        \item \textbf{Full model:} backbone and all heads.
        \item \textbf{Backbone only:} the shared ResNet18 parameters, heads excluded.
      \end{itemize}
      \medskip
      \textbf{Rule:} the reference with the smallest distance is the predicted last task.
    \end{column}
    \begin{column}{0.42\textwidth}
      \textbf{Why two versions?}\\[2pt]
      The full-model distance also includes the heads, which are not comparable in the same way. The backbone version isolates the shared representation.
    \end{column}
  \end{columns}
\end{frame}

% ======================================================
\begin{frame}{Representation similarity: CKA and feature drift}
  Both are computed on the same fixed images. For each image, the 512-dimensional feature vector before the classifier is extracted from the final model and from each reference.

  \medskip
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Linear CKA}
      \[
        \mathrm{CKA}(X,Y)=\frac{\lVert X^\top Y\rVert_F^2}{\lVert X^\top X\rVert_F\,\lVert Y^\top Y\rVert_F}
      \]
      $X,Y$ centred feature matrices.\\
      Asks: is the \emph{structure} of the representation similar?\\
      Range 0 to 1. \hl{Highest CKA} predicts the last task.
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Feature drift}
      \[
        \mathrm{drift}(X,Y)=\operatorname{mean}\left\lVert \tfrac{x}{\lVert x\rVert}-\tfrac{y}{\lVert y\rVert}\right\rVert_2
      \]
      Asks: did the individual feature vectors move?\\
      Range 0 to 2. \hl{Lowest drift} predicts the last task.
    \end{column}
  \end{columns}
\end{frame}

% ======================================================
\begin{frame}{The fixed evaluation image set}
  CKA and feature drift are only comparable if every model sees exactly the same images. Two modes exist:

  \bigskip
  \centering
  \begin{tabular}{@{}lp{3.2cm}p{5.6cm}@{}}
    \toprule
    \textbf{Mode} & \textbf{Images} & \textbf{Details}\\
    \midrule
    \texttt{probe} & 300 & 20 random test images for each of the 15 classes; seed 42; created once and reused\\[3pt]
    \texttt{all\_test} & 1{,}500 & Every CIFAR-100 test image of the 15 classes; no sampling\\
    \bottomrule
  \end{tabular}

  \bigskip
  \raggedright
  The mode is shown in the title of each representation figure. \textbf{The results in this presentation use \texttt{all\_test}.}
\end{frame}

% ======================================================
\begin{frame}{Fisher information: a fingerprint without reference models}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      The Fisher trace measures how strongly the model's output reacts to weight changes on a task's data. It needs only the \hl{final checkpoint}.

      \medskip
      Per task (1{,}000 samples):
      \begin{enumerate}
        \item Sample a label from the model's own predicted probabilities.
        \item Back-propagate the loss; square and accumulate the gradients.
        \item Never update the weights.
      \end{enumerate}
      Result: $F_A$, $F_B$, $F_C$.
    \end{column}
    \begin{column}{0.42\textwidth}
      Nobody knows in advance whether a \emph{high} value means ``recent'' or ``old'', so both readings are tested:
      \medskip

      \small
      \begin{tabular}{@{}ll@{}}
        \toprule
        \textbf{Hypothesis} & \textbf{Last task}\\
        \midrule
        \texttt{high\_is\_recent} & largest $F$\\
        \texttt{low\_is\_recent} & smallest $F$\\
        \bottomrule
      \end{tabular}

      \medskip
      Sorting the three values also gives a guess for the full order (chance: 1/6).
    \end{column}
  \end{columns}
\end{frame}

% ======================================================
\begin{frame}{Weight matching: a control for neuron reordering}
  Two networks can compute the same function with neurons in a different order, which inflates L2 distance. Weight matching (Git Re-Basin) reorders the neurons of one model to best fit the other before measuring.

  \medskip
  \begin{itemize}
    \item Solved with the Hungarian algorithm, one permutation group at a time.
    \item Checked afterwards: the logits of all three heads must be unchanged.
    \item Also computed: cosine similarity of the \emph{weight updates} $\theta_{\text{final}}-\theta_{\text{init}}$ and $\theta_{\text{task}}-\theta_{\text{init}}$.
  \end{itemize}

  \medskip
  All models start from the same initialization, so the expected outcome is the \textbf{identity permutation}: matched and unmatched distances are equal. This makes it a \hl{negative control} for the L2 result.
\end{frame}

% ======================================================
\begin{frame}{Supporting diagnostics (not predictors)}
  \begin{columns}[T]
    \begin{column}{0.32\textwidth}
      \textbf{Forgetting}\\[2pt]
      accuracy right after learning a task minus its final accuracy.\\[2pt]
      Positive: performance dropped.
    \end{column}
    \begin{column}{0.32\textwidth}
      \textbf{Loss barrier}\\[2pt]
      Interpolate linearly between final model and a reference; measure the loss along the path.\\[2pt]
      Height = max minus min loss.
    \end{column}
    \begin{column}{0.32\textwidth}
      \textbf{Jacobian sensitivity}\\[2pt]
      Gradient norm of the predicted logit with respect to the input image, averaged over images.
    \end{column}
  \end{columns}

  \vspace{6mm}
  These describe how the models behave. They do not define a rule for the last task, and higher sensitivity does not mean ``learned more recently''.
\end{frame}

        \item That is by construction: for the last task, ``after learning'' and ``final'' are the same checkpoint.
      \end{itemize}
      So forgetting gives no independent evidence of a fingerprint.
    \end{column}
  \end{columns}
\end{frame}

% ======================================================
\begin{frame}{Limitations}
  \begin{itemize}
    \item \textbf{One training run} (seed 42). No estimate of run-to-run variation.
    \item \textbf{Only six orders.} One flipped prediction changes accuracy by 16.7 points.
    \item \textbf{Fisher predictions are concentrated on A} (4 of 6 orders).
    \item \textbf{Shared initialization.} Weight matching returns the identity, so its benefit cannot show here.
    \item \textbf{Small tasks.} 5 classes each, from CIFAR-100; the tasks may be too similar or too different.
    \item \textbf{No task boundaries or forgetting protection} in training; heads are always 5-class.
    \item \textbf{Hardware.} Seeds give deterministic results on the same hardware only.
  \end{itemize}
\end{frame}

% ======================================================
\begin{frame}{Conclusions and next steps}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Conclusions}
      \begin{itemize}
        \item The last task is detectable in the \hl{representations} (CKA, feature drift) and, less reliably, in the \hl{curvature} (Fisher).
        \item Weight distance and update cosine do not reveal it; both point at the first task.
        \item Forgetting adds no independent evidence.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Next steps}
      \begin{itemize}
        \item Repeat with several training seeds and aggregate.
        \item Use independently initialized references, where weight matching matters.
        \item Find out why weight-space measures favor the first task.
        \item Try other architectures, intermediate layers, other task sets, and more than three tasks.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
